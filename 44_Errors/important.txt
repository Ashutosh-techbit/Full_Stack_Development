4. using wrapAsync function - it wrap a callback and return it in form of function
--> asyncWrap function is a function whose argument is a also a function and it also return a function

fn() - argument

function asyncWrap(fn){
    return function(req,res,next){  //return this function which itself contains fn() function
        fn(req,res,next);
    }
}
