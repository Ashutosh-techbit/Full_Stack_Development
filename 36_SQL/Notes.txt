================WHERE Clause=============
1. Arithmetic operators -> + , - , * , / , %
2. Comparison operators -> = , != , > , < , <= ,>=
3. Logical operators -> AND , OR , NOT , IN , BETWEEN , ALL , LIKE , ANY
4. Bitwise operators -> & , |


select *
from user
where age>25;

================LIMIT=============
//only shows 1 row
select *
from user
where age>25
limit 1;

//======================orderby====================
select * from user
order by followers asc

select * from user
order by followers desc


==========AGGREGATE FUNCTION====================
COUNT() , MIN() , MAX() , AVG() , SUM()


==========groupby====================
select age , count(id)
from user 
group by (age);

=================HAVING================
//having is like where clause but for grouping
select age , max(followers)
from user 
group by (age)
having age <26;
